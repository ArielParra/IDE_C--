main {
    int x, y, z;
    float a, b;
    bool bandera;

    x = 5;
    y = 10;
    z = x + y * 2;
    a = 3.5 + x / 2.0 - 1.0;
    b = (a + 4.0) * 2.0;

    cin >> x;
    cout << "Valor inicial: " << x << "\n";

    if (x > 0 && y <= 20) then
        y = y + x;
        z = z - 1;
    else
        y = y - 1;
        bandera = false;
    end;

    while (z > 0 || !bandera) 
        z = z - 1;
        x++;
    end;

    do
        a = a * 1.1;
        b = b + a;
    while (a < 100.0);

    cout << "Resultado final: " << a << " " << b << "\n";
}
